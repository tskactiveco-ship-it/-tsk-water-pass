CREATE OR REPLACE FUNCTION public.tsk_cashier_order_lookup( p_order_no TEXT ) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$ DECLARE v_branch BIGINT; v_order public.tsk_orders%ROWTYPE; v_member public.tsk_members%ROWTYPE; BEGIN -- Verify staff identity v_branch := public.tsk_scope(); IF p_order_no IS NULL OR length(trim(p_order_no)) < 5 THEN RAISE EXCEPTION 'رقم الطلب غير صحيح'; END IF; SELECT * INTO v_order FROM public.tsk_orders WHERE order_no = trim(p_order_no); IF NOT FOUND THEN RETURN jsonb_build_object( 'success', false, 'message', 'الطلب غير موجود' ); END IF; -- Cashier sees orders for assigned branch only IF v_branch IS NOT NULL AND v_order.pickup_branch_id <> v_branch THEN RAISE EXCEPTION 'هذا الطلب يخص فرعاً آخر'; END IF; SELECT * INTO v_member FROM public.tsk_members WHERE id = v_order.member_id; RETURN jsonb_build_object( 'success', true, 'order_no', v_order.order_no, 'order_type', v_order.order_type, 'member_name', v_member.full_name, 'member_no', v_member.member_no, 'product', COALESCE( v_order.product_name, v_order.product_kind, v_order.catalog_code ), 'quantity', v_order.quantity, 'pickup_branch', v_order.pickup_branch_id, 'payment_method', v_order.payment_method, 'payment_status', v_order.payment_status, 'order_status', v_order.order_status, 'total_amount', v_order.total_amount, 'created_at', v_order.created_at ); END; $$; REVOKE ALL ON FUNCTION public.tsk_cashier_order_lookup(TEXT) FROM PUBLIC, anon, authenticated; GRANT EXECUTE ON FUNCTION public.tsk_cashier_order_lookup(TEXT) TO authenticated;
